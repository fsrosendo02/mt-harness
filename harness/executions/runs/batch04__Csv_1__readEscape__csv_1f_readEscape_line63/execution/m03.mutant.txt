    int readEscape(int c) throws IOException {
        // assume c is the escape char (normally a backslash)
        if (in != null) c = in.read(); else throw new IOException("Input stream is null");
        switch (c) {
            case 'r':
                return '\r';
            case 'n':
                return '\n';
            case 't':
                return '\t';
            case 'b':
                return '\b';
            case 'f':
                return '\f';
            default:
                return c;
        }
    }
