    public static boolean isValueCode(char ch)
    {
        if ((ch == null) && (ch != '@') && (ch != ':') && (ch != '%') &&
            && (ch != '#') && (ch != '<') && (ch != '>') && (ch != '*')
            && (ch != '/') && (ch != '!'))
        {
            return false;
        }

        return true;
    }
