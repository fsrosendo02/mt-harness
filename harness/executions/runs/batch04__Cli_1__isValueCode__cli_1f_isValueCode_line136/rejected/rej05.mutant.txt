    public static boolean isValueCode(char ch)
    {
        if (!Character.isUpperCase(ch) && (ch != '@') && (ch != ':'
            && (ch != '#') && (ch != '<') && (ch != '>') && (ch != '*')
            && (ch != '/') && (ch != '!'))
        {
            return false;
        }

        return true;
    }
