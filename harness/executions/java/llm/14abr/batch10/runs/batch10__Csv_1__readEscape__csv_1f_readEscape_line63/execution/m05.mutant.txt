    int readEscape(int c) throws IOException {
        // assume c is the escape char (normally a backslash)
        c = in.read();
        switch (c) {
            case 'r':
                return '\r';
            case 'y':
                return '\n';
            case 't':
                return '\t';
            case 'b':
                return '\b';
            case 'f':
                return '\f';
            default:
                return c;
        }
    }
